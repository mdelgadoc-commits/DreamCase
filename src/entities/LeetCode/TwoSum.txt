#include <iostream>
#include <vector>
#include <unordered_map>

using namespace std;

vector<int> twoSum(vector<int>& nums, int target) {
    unordered_map<int, int> mapa;
    
    for (int i = 0; i < nums.size(); i++) {
        int complemento = target - nums[i];
        
        if (mapa.find(complemento) != mapa.end()) {
            cout << "Encontrado: " << complemento << " + " << nums[i] << endl;
            cout << "Indices: " << mapa[complemento] << " y " << i << endl;
            return {mapa[complemento], i};
        }
        
        mapa[nums[i]] = i;
    }
    
    cout << "No se encontró solución" << endl;
    return {};
}

int main() {
    cout << "=== Prueba 1 ===" << endl;
    vector<int> nums1 = {1, 2, 3, 4, 5};
    int target1 = 9;
    twoSum(nums1, target1);
    
    cout << "\n=== Prueba 2 ===" << endl;
    vector<int> nums2 = {2, 7, 11, 15};
    int target2 = 9;
    twoSum(nums2, target2);
    
    return 0;
}
